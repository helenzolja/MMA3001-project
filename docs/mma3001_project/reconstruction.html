mma3001_project.reconstruction

Fill missing sensor readings by interpolation and validate the choice of method.

The cleaned series from mma3001_project.load sits on a regular 10-minute grid, where an empty slot is NaN. This module fills those slots from the observed readings on either side (Week 3 interpolation) and never extrapolates: a gap at the very start or end of the record stays NaN.

Methods compared:

  • "locf" - last observation carried forward (baseline)
  • "linear" - piecewise-linear interpolation (3.1 / 3.4)
  • "cubic" - cubic spline (3.4)
  • "pchip" - shape-preserving piecewise cubic Hermite (3.4)

The method is chosen by hiding real readings, filling them and measuring the error (evaluate_methods()).

Example

>>> from mma3001_project.reconstruction import reconstruct_frame
>>> recon = reconstruct_frame(clean, method="linear")

 31def fill_gaps(series, method="linear"):
 32    """Fill the interior gaps of a regularly spaced series.
 33
 34    Parameters
 35    ----------
 36    series : pandas.Series
 37        Series with a DatetimeIndex; ``NaN`` marks a missing reading.
 38    method : {"locf", "linear", "cubic", "pchip"}, default "linear"
 39        Reconstruction method (see the module docstring).
 40
 41    Returns
 42    -------
 43    pandas.Series
 44        Copy of ``series`` with interior gaps filled. Observed readings are
 45        unchanged, and gaps before the first or after the last observation
 46        stay ``NaN`` (no extrapolation).
 47
 48    Raises
 49    ------
 50    ValueError
 51        If ``method`` is unknown or fewer than two readings are observed.
 52    """
 53    if method not in METHODS:
 54        raise ValueError(f"method must be one of {METHODS}, got {method!r}")
 55    y = series.to_numpy(dtype=float)
 56    observed = ~np.isnan(y)
 57    if observed.sum() < 2:
 58        raise ValueError("need at least two observed readings to interpolate")
 59
 60    # Hours since the first slot: a numeric x-axis for the interpolants.
 61    t = ((series.index - series.index[0]).total_seconds() / 3600.0).to_numpy()
 62    t_obs, y_obs = t[observed], y[observed]
 63    missing = ~observed
 64    filled = y.copy()
 65
 66    if method == "locf":
 67        filled = series.ffill().to_numpy(dtype=float, copy=True)
 68    elif method == "linear":
 69        filled[missing] = make_interp_spline(t_obs, y_obs, k=1)(t[missing])
 70    elif method == "cubic":
 71        filled[missing] = CubicSpline(t_obs, y_obs)(t[missing])
 72    elif method == "pchip":
 73        filled[missing] = PchipInterpolator(t_obs, y_obs)(t[missing])
 74
 75    # Interpolation only: blank anything outside the observed time range.
 76    outside = (t < t_obs[0]) | (t > t_obs[-1])
 77    filled[missing & outside] = np.nan
 78    return pd.Series(filled, index=series.index, name=series.name)

 81def reconstruct_frame(frame, method="linear"):
 82    """Fill every column of a cleaned frame and flag what was reconstructed.
 83
 84    Parameters
 85    ----------
 86    frame : pandas.DataFrame
 87        Output of :func:`mma3001_project.load.clean_sensor_frame`.
 88    method : str, default "linear"
 89        Passed to :func:`fill_gaps`.
 90
 91    Returns
 92    -------
 93    pandas.DataFrame
 94        The filled columns plus a boolean ``<column>_reconstructed`` column
 95        for each, True where the value was filled rather than measured.
 96    """
 97    out = pd.DataFrame(index=frame.index)
 98    for col in frame.columns:
 99        filled = fill_gaps(frame[col], method=method)
100        out[col] = filled
101        out[f"{col}_reconstructed"] = frame[col].isna() & filled.notna()
102    return out

105def sample_gap_starts(series, length, n, rng):
106    """Choose random positions for artificial gaps of ``length`` slots.
107
108    A start ``s`` is allowed only if every hidden slot holds a real reading
109    (so there is a true value to score against) and the slots just before
110    (``s-1``) and after (``s+length``) are observed, like a real gap.
111
112    Parameters
113    ----------
114    series : pandas.Series
115        Regularly spaced series; ``NaN`` marks a genuinely missing reading.
116    length : int
117        Gap length in slots (1 slot = 10 minutes).
118    n : int
119        Number of gaps to choose.
120    rng : numpy.random.Generator
121        Seeded random generator, so results are reproducible.
122
123    Returns
124    -------
125    numpy.ndarray
126        Integer start positions (without repeats when enough are available).
127
128    Raises
129    ------
130    ValueError
131        If no allowed start position exists.
132    """
133    ok = series.notna().to_numpy()
134    counts = np.concatenate(([0], np.cumsum(ok)))     # running count of readings
135    s = np.arange(1, len(ok) - length)                # needs s-1 and s+length
136    full = (counts[s + length] - counts[s]) == length # every hidden slot observed
137    starts = s[full & ok[s - 1] & ok[s + length]]
138    if len(starts) == 0:
139        raise ValueError(f"no allowed start positions for a gap of {length} slots")
140    return rng.choice(starts, size=n, replace=len(starts) < n)

143def evaluate_methods(series, lengths, n_gaps=200, seed=0, methods=METHODS, window=72):
144    """Hide real readings, fill them with each method, record the errors.
145
146    For every gap length, ``n_gaps`` artificial gaps are placed at random.
147    Each method fills the gap from the surrounding ``window`` slots on each
148    side, and the error (filled - true) is recorded at every hidden slot.
149
150    Parameters
151    ----------
152    series : pandas.Series
153        Regularly spaced series with a DatetimeIndex.
154    lengths : list of int
155        Gap lengths to test, in slots (1 slot = 10 minutes).
156    n_gaps : int, default 200
157        Number of artificial gaps per length.
158    seed : int, default 0
159        Random seed (fixed, so the experiment is reproducible).
160    methods : sequence of str, default METHODS
161        Methods passed to :func:`fill_gaps`.
162    window : int, default 72
163        Slots of context kept on each side of a gap (72 = 12 hours).
164
165    Returns
166    -------
167    pandas.DataFrame
168        One row per (method, hidden reading) with columns ``method``,
169        ``length``, ``gap_id`` and ``error``.
170    """
171    rng = np.random.default_rng(seed)
172    values = series.to_numpy(dtype=float)
173    rows = []
174    gap_id = 0
175    for length in lengths:
176        for s in sample_gap_starts(series, length, n_gaps, rng):
177            # Local window keeps each fill O(window), not O(series); see 07_performance.
178            lo, hi = max(0, s - window), min(len(series), s + length + window)
179            local = series.iloc[lo:hi].copy()
180            local.iloc[s - lo:s - lo + length] = np.nan       # hide the readings
181            truth = values[s:s + length]
182            for m in methods:
183                pred = fill_gaps(local, method=m).to_numpy()[s - lo:s - lo + length]
184                rows += [(m, length, gap_id, e) for e in pred - truth]
185            gap_id += 1
186    return pd.DataFrame(rows, columns=["method", "length", "gap_id", "error"])

189def summarise_errors(errors, by=("method", "length")):
190    """Pooled RMSE, MAE and bias for each group of an :func:`evaluate_methods` table.
191
192    Parameters
193    ----------
194    errors : pandas.DataFrame
195        Output of :func:`evaluate_methods`.
196    by : sequence of str, default ("method", "length")
197        Columns to group by.
198
199    Returns
200    -------
201    pandas.DataFrame
202        Columns ``RMSE``, ``MAE``, ``bias`` (same units as the data) and
203        ``n`` for each group.
204    """
205    g = errors.groupby(list(by))["error"]
206    return pd.DataFrame({
207        "RMSE": np.sqrt(g.apply(lambda e: np.mean(e ** 2))),
208        "MAE": g.apply(lambda e: np.mean(np.abs(e))),
209        "bias": g.mean(),
210        "n": g.size(),
211    })

214def bootstrap_rmse(errors, n_boot=2000, seed=0, level=0.95):
215    """Bootstrap confidence interval for a pooled RMSE, resampling whole gaps.
216
217    Readings inside one gap are not independent, so whole gaps (not single
218    readings) are resampled with replacement (Week 5.5.11).
219
220    Parameters
221    ----------
222    errors : pandas.DataFrame
223        Rows of :func:`evaluate_methods` for ONE method.
224    n_boot : int, default 2000
225        Number of bootstrap resamples.
226    seed : int, default 0
227        Random seed.
228    level : float, default 0.95
229        Confidence level.
230
231    Returns
232    -------
233    tuple of float
234        (RMSE, lower bound, upper bound).
235    """
236    # Reduce each gap to (sum of squared errors, count) so a resample is a cheap sum.
237    per_gap = errors.groupby("gap_id")["error"].agg(
238        sse=lambda e: np.sum(e ** 2), n="size")
239    sse, n = per_gap["sse"].to_numpy(), per_gap["n"].to_numpy()
240    rng = np.random.default_rng(seed)
241    pick = rng.integers(0, len(sse), size=(n_boot, len(sse)))   # gap indices per resample
242    boot = np.sqrt(sse[pick].sum(axis=1) / n[pick].sum(axis=1))
243    alpha = (1 - level) / 2
244    return (float(np.sqrt(sse.sum() / n.sum())), float(np.quantile(boot, alpha)), float(np.quantile(boot, 1 - alpha)))
